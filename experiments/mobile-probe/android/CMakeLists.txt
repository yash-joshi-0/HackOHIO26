cmake_minimum_required(VERSION 3.28)
project(MeshThingAndroidProbe LANGUAGES C CXX)

set(MESH_LLAMA_ROOT "" CACHE PATH "Private patched llama.cpp clone")
if(NOT EXISTS "${MESH_LLAMA_ROOT}/include/llama.h")
  message(FATAL_ERROR "Set MESH_LLAMA_ROOT to the patched runtime")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
add_subdirectory("${MESH_LLAMA_ROOT}" runtime)
add_executable(mesh-native-smoke main.cpp ../native_smoke.cpp)
target_compile_features(mesh-native-smoke PRIVATE cxx_std_17)
target_include_directories(mesh-native-smoke PRIVATE "${MESH_LLAMA_ROOT}/src")
target_link_libraries(mesh-native-smoke PRIVATE llama)
add_executable(mesh-shard-socket socket_main.cpp runner_impl.cpp)
target_compile_features(mesh-shard-socket PRIVATE cxx_std_17)
target_include_directories(mesh-shard-socket PRIVATE "${MESH_LLAMA_ROOT}/src")
target_link_libraries(mesh-shard-socket PRIVATE llama)
